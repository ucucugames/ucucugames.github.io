using UnityEngine; using System.Collections.Generic; public class CarController : MonoBehaviour { // Tekerleklerin fiziksel bileşenleri (Wheel Collider) public WheelCollider onSol, onSag, arkaSol, arkaSag; // Tekerleklerin görsel bileşenleri (Tekerlek modelleri) public Transform onSolModel, onSagModel, arkaSolModel, arkaSagModel; public float motorGucu = 2000f; public float frenGucu = 3000f; public float maxDonusAcisi = 35f; private float gazInput; private float direksiyonInput; private bool frenYapiliyor mu; void Update() { // Klavyeden girişleri al (W,A,S,D veya Ok Tuşları) gazInput = Input.GetAxis("Vertical"); direksiyonInput = Input.GetAxis("Horizontal"); frenYapiliyor mu = Input.GetKey(KeyCode.Space); TekerlekModelleriniGuncelle(); } void FixedUpdate() { // 1. Motor Gücü Uygula arkaSol.motorTorque = gazInput * motorGucu; arkaSag.motorTorque = gazInput * motorGucu; // 2. Direksiyon Dönüşü onSol.steerAngle = direksiyonInput * maxDonusAcisi; onSag.steerAngle = direksiyonInput * maxDonusAcisi; // 3. Fren Sistemi if (frenYapiliyor mu) { onSol.brakeTorque = frenGucu; onSag.brakeTorque = frenGucu; arkaSol.brakeTorque = frenGucu; arkaSag.brakeTorque = frenGucu; } else { onSol.brakeTorque = 0; onSag.brakeTorque = 0; arkaSol.brakeTorque = 0; arkaSag.brakeTorque = 0; } } // Tekerlek modellerinin (Görselin) Collider ile aynı hizada dönmesini sağlar void TekerlekModelleriniGuncelle() { TekerlekPozisyonuGuncelle(onSol, onSolModel); TekerlekPozisyonuGuncelle(onSag, onSagModel); TekerlekPozisyonuGuncelle(arkaSol, arkaSolModel); TekerlekPozisyonuGuncelle(arkaSag, arkaSagModel); } void TekerlekPozisyonuGuncelle(WheelCollider collider, Transform model) { Vector3 pos; Quaternion rot; collider.GetWorldPose(out pos, out rot); model.position = pos; model.rotation = rot; } }
